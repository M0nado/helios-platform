cmake_minimum_required(VERSION 3.25)
project(HeliosPreviewRenderer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(helios_preview_renderer renderer.cpp)
if(MSVC)
  target_compile_options(helios_preview_renderer PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(helios_preview_renderer PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
